#
# Python Binding Generation
#

if (NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Choose the type of build." FORCE)
  set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS "Debug" "Release" "MinSizeRel" "RelWithDebInfo")
endif()

#
# Define the Module and Bindings
#

find_package(Python 3.8 COMPONENTS Interpreter Development.Module REQUIRED)
if(NOT TARGET nanobind::nanobind_static AND NOT TARGET nanobind)
  add_subdirectory(${CMAKE_SOURCE_DIR}/ext/nanobind nanobind)
endif()

set(MODULE_NAME silt)
nanobind_add_module(${MODULE_NAME}
  NB_SHARED
  NB_DOMAIN "erosiv"
  source/_module.cpp
  source/shape.cpp
  source/slice.cpp
  source/tensor.cpp
  source/view.cpp
  source/op.cpp
  source/util.cpp
  source/indexed.cpp
  source/rng.cpp
)

# Linux RPATH configuration
set_target_properties(${MODULE_NAME} PROPERTIES
  INSTALL_RPATH "$ORIGIN"
  BUILD_WITH_INSTALL_RPATH FALSE
)

target_compile_definitions(${MODULE_NAME} PRIVATE
  MODULE_NAME=${MODULE_NAME})

#
# Dependencies
#

#
# Type Stubs (.pyi)
#
#  python/silt/{__init__,silt}.pyi are checked in and regenerated by every
#  build, so they always describe this build of the extension. CI fails when
#  the committed stubs differ from the regenerated ones.
#

option(SILT_UPDATE_STUBS "Regenerate python/silt/*.pyi after building the extension" ON)

set(STUB_STAGE ${CMAKE_CURRENT_BINARY_DIR}/stub_stage)
set(STUB_PACKAGE ${CMAKE_CURRENT_SOURCE_DIR}/silt)

# Runtime DLLs beside the extension so the staged package imports on Windows.
set(STAGE_DLLS "")
if(WIN32)
  set(STAGE_DLLS COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_RUNTIME_DLLS:${MODULE_NAME}>" ${STUB_STAGE}/silt)
endif()

# Stage the package with the freshly built extension so stubgen imports it, not an installed copy.
add_custom_command(TARGET ${MODULE_NAME} POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory ${STUB_STAGE}/silt
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${MODULE_NAME}>" ${STUB_STAGE}/silt
  ${STAGE_DLLS}
  COMMAND_EXPAND_LISTS
)

add_custom_command(
  OUTPUT ${STUB_STAGE}/silt/__init__.py
  COMMAND ${CMAKE_COMMAND} -E make_directory ${STUB_STAGE}/silt
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${STUB_PACKAGE}/__init__.py ${STUB_STAGE}/silt
  DEPENDS ${STUB_PACKAGE}/__init__.py
)

if(SILT_UPDATE_STUBS)
  # INCLUDE_PRIVATE: stubgen otherwise drops every trailing-underscore name, i.e. all in-place ops.
  nanobind_add_stub(
    ${MODULE_NAME}_stub
    MODULE ${MODULE_NAME}
    RECURSIVE
    INCLUDE_PRIVATE
    PYTHON_PATH ${STUB_STAGE}
    OUTPUT_PATH ${STUB_PACKAGE}
    OUTPUT ${STUB_PACKAGE}/__init__.pyi ${STUB_PACKAGE}/${MODULE_NAME}.pyi
    DEPENDS ${MODULE_NAME} ${STUB_STAGE}/silt/__init__.py
  )
endif()

if(MSVC)
add_custom_command(TARGET ${MODULE_NAME} POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_RUNTIME_DLLS:${MODULE_NAME}>"
    "$<TARGET_FILE_DIR:${MODULE_NAME}>"
  COMMAND_EXPAND_LISTS
)
endif()

# glm::glm transitively linked through silt
target_link_libraries(${MODULE_NAME} PRIVATE silt::silt)

#
# Installation
#

# Install Directive for scikit-build module
install(TARGETS ${MODULE_NAME} LIBRARY DESTINATION ${MODULE_NAME})

install(TARGETS silt_lib
  RUNTIME DESTINATION ${SKBUILD_PLATLIB_DIR}/${MODULE_NAME} # Windows
  LIBRARY DESTINATION ${SKBUILD_PLATLIB_DIR}/${MODULE_NAME} # Linux
)

install(TARGETS nanobind-erosiv
  RUNTIME DESTINATION ${SKBUILD_PLATLIB_DIR}/${MODULE_NAME} # Windows
  LIBRARY DESTINATION ${SKBUILD_PLATLIB_DIR}/${MODULE_NAME} # Linux
)
